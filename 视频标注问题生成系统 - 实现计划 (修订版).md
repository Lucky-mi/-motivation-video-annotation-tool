## 项目目标

为现有的AI视频标注系统添加**问题生成与校验功能**,用于评估多模态大模型在推理motivation和desire方面的能力。

### 核心功能

1. **AI自动生成单选题** - 基于已标注的视频关键帧,生成关于desire的单选题
2. **人工校验界面(优先)** - 提供流畅的问题校验页面,支持编辑和审核
3. **帧切换优化** - 优化关键帧切换性能,实现预加载和缓存
4. **研究数据导出** - 支持导出问题集和校验结果

---

## 设计原则

IMPORTANT

**核心设计原则 - 基于用户反馈:**

1. **低耦合架构** - 问题生成逻辑与prompt模板完全分离
    - Prompt存储在独立的YAML配置文件中
    - 支持运行时热加载,无需重启服务
    - 便于后续迭代和优化
2. **灵活的标签体系** - desire标签暂不固化
    - 数据模型使用灵活的dict字段
    - 支持后续根据标注数据总结标签
    - 不预设固定的desire分类
3. **简化问题类型** - 暂时不细分复杂类型
    - 主要问题: "推理desire是什么"
    - 辅助问题: "为什么是这个desire"
    - 后续可根据需要扩展
4. **校验优先** - 先实现校验模块
    - 问题展示和编辑界面优先开发
    - 生成逻辑可后续迭代优化
    - 确保人工校验体验流畅

class PromptLoader:

    """Prompt模板加载器 - 支持热加载"""

    def __init__(self, config_path: str = "backend/prompts/question_prompts.yaml"):

        self.config_path = config_path

        self.prompts = {}

        self.load_prompts()

    def load_prompts(self):

        """加载prompt配置"""

        with open(self.config_path, 'r', encoding='utf-8') as f:

            self.prompts = yaml.safe_load(f)

    def reload(self):

        """热加载 - 无需重启服务"""

        self.load_prompts()

    def get_prompt(self, prompt_key: str, **kwargs) -> str:

        """获取并格式化prompt"""

        template = self.prompts.get(prompt_key, "")

        return template.format(**kwargs)

class QuestionGenerator:

    """问题生成器 - 完全解耦"""

    def __init__(self, ai_provider, prompt_loader: PromptLoader = None):

        self.ai_provider = ai_provider

        self.prompt_loader = prompt_loader or PromptLoader()

    def generate_questions_for_video(

        self,

        video_id: str,

        annotation_data: VideoAnnotation,

        num_questions: int = 10

    ) -> QuestionSet:

        """为视频生成问题集"""

        questions = []

        # 从标注数据中选择关键帧

        selected_frames = self._select_frames_for_questions(

            annotation_data.keyframes,

            num_questions

        )

        for frame in selected_frames:

            # 生成"desire是什么"类型的问题

            question = self._generate_desire_inference_question(frame)

            questions.append(question)

        return QuestionSet(

            question_set_id=str(uuid.uuid4()),

            video_id=video_id,

            video_name=annotation_data.video_name,

            questions=questions,

            total_questions=len(questions),

            verified_questions=0,

            created_at=datetime.now().isoformat(),

            last_modified=datetime.now().isoformat(),

            ai_provider=self.ai_provider.name

        )

    def _generate_desire_inference_question(

        self,

        keyframe: KeyframeAnnotation

    ) -> Question:

        """生成desire推理问题"""

        # 从prompt配置加载模板

        prompt = self.prompt_loader.get_prompt(

            "desire_inference",

            action_description=keyframe.action.action_description,

            visual_context=keyframe.action.visual_context,

            timestamp=keyframe.action.timestamp_formatted

        )

        # 调用AI生成问题和选项

        response = self.ai_provider.generate_question(prompt)

        # 解析响应

        question_data = json.loads(response)

        return Question(

            question_id=str(uuid.uuid4()),

            question_text=question_data['question_text'],

            options=[

                QuestionOption(**opt) for opt in question_data['options']

            ],

            correct_option_id=question_data['correct_option_id'],

            related_frame_ids=[keyframe.frame_id],

            reasoning_process=question_data.get('reasoning', ''),

            created_at=datetime.now().isoformat()

        )

@app.post("/questions/generate/{video_id}")

async def generate_questions(

    video_id: str,

    num_questions: int = Query(default=10, ge=1, le=50),

    provider: str = Query(default="gemini")

):

    """为视频生成问题集"""

    try:

        # 加载标注数据

        annotation_path = ANNOTATIONS_DIR / f"{video_id}.json"

        with open(annotation_path, 'r', encoding='utf-8') as f:

            annotation_data = VideoAnnotation(**json.load(f))

        # 获取AI provider

        ai_provider = AIProviderFactory.get_provider(provider)

        # 生成问题

        generator = QuestionGenerator(ai_provider, prompt_loader)

        question_set = generator.generate_questions_for_video(

            video_id,

            annotation_data,

            num_questions

        )

        # 保存问题集

        questions_dir = Path("data/questions")

        questions_dir.mkdir(exist_ok=True)

        output_path = questions_dir / f"{video_id}_questions.json"

        with open(output_path, 'w', encoding='utf-8') as f:

            json.dump(question_set.dict(), f, ensure_ascii=False, indent=2)

        return {

            "success": True,

            "question_set_id": question_set.question_set_id,

            "total_questions": question_set.total_questions,

            "message": f"成功生成{question_set.total_questions}个问题"

        }

    except Exception as e:

        raise HTTPException(status_code=500, detail=str(e))
