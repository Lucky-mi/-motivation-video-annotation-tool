```python
async def batch_check(
    self,
    video_paths: list,
    strict_mode: bool = True,
    max_concurrent: int = 5
) -> Dict[str, Dict]:
    """
    批量检查多个视频 (并发加速版)

    Args:
        video_paths: 视频路径列表
        strict_mode: 严格模式
        max_concurrent: 最大并发数

    Returns:
        {video_path: review_result} 字典
    """
    results = {}
    total = len(video_paths)
    logger.info(f"🚀 开始批量审核 {total} 个视频 (并发数: {max_concurrent})...")

    # 使用信号量控制并发
    semaphore = asyncio.Semaphore(max_concurrent)

    async def check_one(path):
        async with semaphore:
            try:
                logger.info(f"➡️ 开始审核: {Path(path).name}")
                result = await self.check_video_content(path, strict_mode)
                status = "✅" if result.get('pass') else "❌"
                logger.info(f"{status} 完成: {Path(path).name}")
                return path, result
            except Exception as e:
                logger.error(f"❌ 审核异常 {Path(path).name}: {e}")
                return path, {"pass": False, "reason": f"异常: {str(e)}", "confidence": 0.0}

    # 并发执行所有任务
    tasks = [check_one(path) for path in video_paths]
    completed = await asyncio.gather(*tasks, return_exceptions=True)

    # 整理结果
    for item in completed:
        if isinstance(item, Exception):
            logger.error(f"任务异常: {item}")
            continue
        path, result = item
        results[path] = result

    # 统计结果
    passed = sum(1 for r in results.values() if r.get("pass", False))
    logger.info(f"🏁 批量审核完成: {passed}/{total} 个视频通过")

    return results
```

### 5. 生产者-消费者流水线模式

**当前问题**: 单视频内下载和审核串行,导致资源利用率不高

**优化方案**: 使用队列解耦下载和审核

**新增文件**: `scripts/run_async_pipeline.py`
```python
#!/usr/bin/env python
"""
改进的流水线模式 - 使用队列实现真正的流水线并行
"""
import sys, os
from pathlib import Path
if sys.platform == 'win32':
    os.system('chcp 65001 > nul')
    sys.stdout.reconfigure(encoding='utf-8')
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))
import asyncio, json, time, argparse, logging
sys.path.insert(0, str(project_root / "config"))
from search_config import ENABLE_AI_REVIEW, STRICT_MODE, AUTO_DELETE_REJECTED, AI_REVIEW_WORKERS
from backend.downloader import VideoDownloader
from backend.content_filter import ContentFilter
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class PipelineVideoProcessor:
    """流水线模式的视频处理器"""

    def __init__(self, downloader, content_filter=None,
                 download_workers=2, review_workers=3):
        self.downloader = downloader
        self.content_filter = content_filter
        self.download_workers = download_workers
        self.review_workers = review_workers
        self.stats = {
            'downloaded': 0,
            'download_failed': 0,
            'reviewed': 0,
            'approved': 0,
            'rejected': 0,
            'skipped': 0,
            'review_failed': 0
        }

    async def download_worker(self, download_queue, review_queue, worker_id):
        """下载工作协程"""
        while True:
            item = await download_queue.get()
            if item is None:  # 结束信号
                download_queue.task_done()
                break

            video_info, index, total = item
            try:
                logger.info(f"[Worker-{worker_id}] [{index}/{total}] Downloading...")
                loop = asyncio.get_event_loop()
                result = await loop.run_in_executor(
                    None,
                    self.downloader.download_from_url,
                    video_info['url']
                )

                if result.get('skipped', False):
                    self.stats['skipped'] += 1
                else:
                    self.stats['downloaded'] += 1
                    result['video_info'] = video_info
                    result['index'] = index
                    result['total'] = total
                    # 放入审核队列
                    await review_queue.put(result)

            except Exception as e:
                self.stats['download_failed'] += 1
                logger.error(
                    f"[Worker-{worker_id}] [{index}/{total}] Download failed: {e}",
                    exc_info=True
                )

            download_queue.task_done()

    async def review_worker(self, review_queue, worker_id):
        """审核工作协程"""
        while True:
            item = await review_queue.get()
            if item is None:  # 结束信号
                review_queue.task_done()
                break

            video_path = item['video_path']
            video_info = item['video_info']
            index = item['index']
            total = item['total']

            try:
                logger.info(f"[Reviewer-{worker_id}] [{index}/{total}] Reviewing...")
                review_result = await self.content_filter.check_video_content(
                    video_path,
                    strict_mode=STRICT_MODE
                )

                self.stats['reviewed'] += 1
                approved = review_result.get('pass', False)

                if approved:
                    self.stats['approved'] += 1
                else:
                    self.stats['rejected'] += 1

                # 保存结果
                self.downloader.add_video_link(
                    url=video_info['url'],
                    title=video_info['title'],
                    duration=item['duration'],
                    keyword=video_info.get('search_keyword', 'Unknown'),
                    approved=approved,
                    review_reason=review_result.get('reason', ''),
                    video_path=video_path
                )

                # 删除未通过的视频
                if not approved and AUTO_DELETE_REJECTED:
                    Path(video_path).unlink(missing_ok=True)

            except Exception as e:
                self.stats['review_failed'] += 1
                logger.error(
                    f"[Reviewer-{worker_id}] [{index}/{total}] Review failed: {e}",
                    exc_info=True
                )

            review_queue.task_done()

    async def process_batch(self, videos):
        """批量处理视频 - 流水线模式"""
        total = len(videos)
        download_queue = asyncio.Queue()
        review_queue = asyncio.Queue()

        # 启动下载工作协程
        download_tasks = [
            asyncio.create_task(
                self.download_worker(download_queue, review_queue, i+1)
            )
            for i in range(self.download_workers)
        ]

        # 启动审核工作协程 (如果启用)
        review_tasks = []
        if self.content_filter:
            review_tasks = [
                asyncio.create_task(
                    self.review_worker(review_queue, i+1)
                )
                for i in range(self.review_workers)
            ]

        # 填充下载任务
        for idx, video in enumerate(videos):
            await download_queue.put((video, idx + 1, total))

        # 等待所有下载完成
        await download_queue.join()

        # 发送下载结束信号
        for _ in range(self.download_workers):
            await download_queue.put(None)

        # 等待下载工作协程结束
        await asyncio.gather(*download_tasks)

        if self.content_filter:
            # 等待所有审核完成
            await review_queue.join()

            # 发送审核结束信号
            for _ in range(self.review_workers):
                await review_queue.put(None)

            # 等待审核工作协程结束
            await asyncio.gather(*review_tasks)

    def print_stats(self):
        print("\nStatistics:")
        print(f"Downloaded: {self.stats['downloaded']} | "
              f"Skipped: {self.stats['skipped']} | "
              f"Failed: {self.stats['download_failed']}")
        if self.content_filter:
            print(f"Reviewed: {self.stats['reviewed']} | "
                  f"Approved: {self.stats['approved']} | "
                  f"Rejected: {self.stats['rejected']} | "
                  f"Failed: {self.stats['review_failed']}")
            if self.stats['reviewed'] > 0:
                approval_rate = self.stats['approved'] / self.stats['reviewed'] * 100
                print(f"Approval Rate: {approval_rate:.1f}%")


def load_search_results(file_path):
    path = Path(file_path)
    if not path.exists():
        return []
    with open(path, 'r', encoding='utf-8') as f:
        data = json.load(f)
    return data if isinstance(data, list) else data.get('videos', [])


async def main_async(args):
    print("Pipeline Video Processing System")
    videos = load_search_results(args.input)
    if not videos:
        print("No videos found!")
        return

    downloader = VideoDownloader(proxy=args.proxy)
    content_filter = ContentFilter() if ENABLE_AI_REVIEW and not args.skip_review else None

    # 过滤已存在的视频
    existing = {v['url'] for v in downloader.links_db['videos']}
    videos = [v for v in videos if v['url'] not in existing]
    videos = videos[args.start:args.start + args.limit if args.limit else None]

    print(f"Config: Download Workers={args.download_workers}, "
          f"Review Workers={args.review_workers}, Videos={len(videos)}")

    if not args.yes and input("Continue? (y/n): ").lower() != 'y':
        return

    processor = PipelineVideoProcessor(
        downloader,
        content_filter,
        args.download_workers,
        args.review_workers
    )

    print("Starting pipeline processing...")
    start = time.time()
    await processor.process_batch(videos)
    duration = time.time() - start

    processor.print_stats()
    print(f"Total time: {duration:.1f}s")
    print(f"Throughput: {len(videos) / duration * 60:.2f} videos/hour")


def main():
    p = argparse.ArgumentParser()
    p.add_argument('--input', default='data/search_results.json')
    p.add_argument('--start', type=int, default=0)
    p.add_argument('--limit', type=int)
    p.add_argument('--download-workers', type=int, default=2)
    p.add_argument('--review-workers', type=int, default=AI_REVIEW_WORKERS)
    p.add_argument('--skip-review', action='store_true')
    p.add_argument('--proxy')
    p.add_argument('-y', '--yes', action='store_true')
    asyncio.run(main_async(p.parse_args()))


if __name__ == '__main__':
    main()
```

### 7. 动态调整并发数

**新增功能**: 根据错误率动态调整并发

```python
class AdaptiveVideoProcessor(AsyncVideoProcessor):
    """自适应并发的处理器"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.consecutive_errors = 0
        self.max_errors_threshold = 5

    async def download_video(self, video_info, index, total):
        try:
            result = await super().download_video(video_info, index, total)
            # 成功后重置错误计数
            self.consecutive_errors = max(0, self.consecutive_errors - 1)
            return result
        except Exception as e:
            self.consecutive_errors += 1

            # 如果连续错误过多,降低并发
            if self.consecutive_errors >= self.max_errors_threshold:
                logger.warning(f"⚠️ 检测到连续 {self.consecutive_errors} 个错误,降低并发...")
                await self._reduce_concurrency()

            raise

    async def _reduce_concurrency(self):
        """降低并发数"""
        # 释放一些信号量许可
        current_download = self.download_sem._value
        current_review = self.review_sem._value

        if current_download > 1:
            # 创建新的更小的信号量
            new_download = max(1, current_download - 1)
            self.download_sem = asyncio.Semaphore(new_download)
            logger.info(f"  📉 下载并发: {current_download} → {new_download}")

        if current_review > 1:
            new_review = max(1, current_review - 1)
            self.review_sem = asyncio.Semaphore(new_review)
            logger.info(f"  📉 审核并发: {current_review} → {new_review}")

        # 等待一段时间
        await asyncio.sleep(5)
        self.consecutive_errors = 0
```

class JSONFormatter(logging.Formatter):
    """JSON 格式的日志"""

    def format(self, record):
        log_obj = {

            "level": record.levelname,

from dataclasses import dataclass, field
from typing import List
import time


@dataclass
class PerformanceMetrics:
    """性能指标收集"""
    download_times: List[float] = field(default_factory=list)
    review_times: List[float] = field(default_factory=list)
    total_start_time: float = 0

    def record_download(self, duration: float):
        self.download_times.append(duration)

    def record_review(self, duration: float):
        self.review_times.append(duration)

    def start(self):
        self.total_start_time = time.time()

            "avg_review_time": sum(self.review_times) / len(self.review_times) if self.review_times else 0,
            "throughput_per_hour": len(self.download_times) / total_time * 3600 if total_time > 0 else 0
        }

# 异步视频处理系统使用指南

## 快速开始

## 参数说明

- `--download-workers`: 下载并发数 (推荐: Windows=2, Linux=3)
- `--review-workers`: 审核并发数 (推荐: Windows=3, Linux=5)
- `--limit`: 处理视频数量限制
- `--proxy`: 代理地址
- `--skip-review`: 跳过 AI 审核

## 故障排查

### WinError 10055

**原因**: Windows 套接字资源耗尽

**解决**:
1. 降低并发数: `--download-workers 1 --review-workers 2`
2. 增加延迟: 修改 `search_config.py` 中的速率限制

### API 限流

**原因**: 请求过快

**解决**:
1. 降低审核并发: `--review-workers 2`
